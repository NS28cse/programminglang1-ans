# 第7回 課題3 LineInput
# 本体は講義の line.c（容量 32）。「容量を小さくして仕組みを見る」（line[32] を line[8] に変える）版をテストする
# 期待値は variants/tests/line8--<入力の名前>.*。容量に関するコメントも 8（本文 7 バイト）に合わせる
softprac_add_program(LineInput line.c)

softprac_add_variant(LineInput line.c line8
  "// 講義の line.c。1 行を char line[32] へ読み込み，本文の長さと内容を表示する。"
  "// 講義の line.c の容量を char line[8] に小さくした版。1 行を読み込み，本文の長さと内容を表示する。"
  "// 本文が 31 バイトを超えた行と"
  "// 本文が 7 バイトを超えた行と"
  [=[    char line[32]; // 本文は最大 31 バイト＋終端（改行まで入るのは本文 30 バイト以下）]=]
  [=[    char line[8]; // 本文は最大 7 バイト＋終端（改行まで入るのは本文 6 バイト以下）]=]
  [=[        // 改行が入らなかった: ちょうど 31 バイトか，超過か，改行なしで終了したか]=]
  [=[        // 改行が入らなかった: ちょうど 7 バイトか，超過か，改行なしで終了したか]=])
